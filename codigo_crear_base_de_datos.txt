-- 1. Crear y usar la base de datos
CREATE DATABASE IF NOT EXISTS junta_vecinal_db;
USE junta_vecinal_db;

-- 2. Crear tabla central Persona
CREATE TABLE Persona (
    rut VARCHAR(12) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL,
    rol ENUM('Admin', 'Presidente', 'Secretario', 'Tesorero', 'Vecino') NOT NULL
);

-- 3. Crear tabla Vecino_Socio (Hereda de Persona)
CREATE TABLE Vecino_Socio (
    id INT AUTO_INCREMENT PRIMARY KEY,
    rut_persona VARCHAR(12),
    direccion VARCHAR(200),
    fecha_ingreso DATE,
    FOREIGN KEY (rut_persona) REFERENCES Persona(rut) ON DELETE CASCADE
);

-- 4. Crear tabla Dirigente (Hereda de Persona)
CREATE TABLE Dirigente (
    id INT AUTO_INCREMENT PRIMARY KEY,
    rut_persona VARCHAR(12),
    cargo ENUM('Presidente', 'Secretario', 'Tesorero', 'Admin'),
    fecha_asuncion DATE,
    FOREIGN KEY (rut_persona) REFERENCES Persona(rut) ON DELETE CASCADE
);

-- 5. Crear tabla Certificado
CREATE TABLE Certificado (
    id INT AUTO_INCREMENT PRIMARY KEY,
    rut_persona VARCHAR(12),
    tipo_certificado VARCHAR(100),
    fecha_emision DATE,
    FOREIGN KEY (rut_persona) REFERENCES Persona(rut) ON DELETE CASCADE
);
